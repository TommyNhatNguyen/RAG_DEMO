\chapter{THỰC NGHIỆM, ĐÁNH GIÁ VÀ THẢO LUẬN}
\label{Chapter4}

\section*{Tóm tắt chương}

Chương này trình bày quá trình thực nghiệm bao gồm: môi trường thực nghiệm, dữ liệu, cấu hình của ba hệ thống truy xuất được so sánh, kết quả truy xuất kèm kiểm định thống kê, và kết quả đánh giá chất lượng sinh câu trả lời (độ trung thực với ngữ cảnh, độ liên quan của câu trả lời, độ chính xác và độ phủ của trích dẫn). Kho học liệu dùng trong thực nghiệm đạt độ phủ 100\% đối với bộ câu hỏi đánh giá, và cấu hình lai đa phương thức có xếp hạng lại cho kết quả cao hơn hệ thống cơ sở với khoảng tin cậy lấy mẫu lại không chứa 0 ở phần lớn chỉ số tại $K=5$ và $K=10$ — bằng chứng thống kê cho thấy cấu hình đề xuất vượt cơ sở. Chương cũng ghi rõ trạng thái của các thí nghiệm: đã có kết quả, và sẽ thực hiện, để người đọc phân biệt được đâu là kết quả đã kiểm chứng và đâu là kế hoạch.

\section{Mục tiêu và phạm vi của thực nghiệm}

% [LOG EXPERIMENT_LOG.md mục 16]
Thực nghiệm nhằm trả lời hai câu hỏi. Thứ nhất, với cùng một bộ câu hỏi về học liệu của ba môn học, các thành phần truy xuất bổ sung (BM25, truy xuất trên không gian thị giác, bộ xếp hạng lại) có cải thiện khả năng tìm đúng bằng chứng so với truy xuất dày chỉ trên văn bản hay không, và việc định dạng đầu vào cho bộ xếp hạng lại đúng theo mẫu hội thoại của mô hình có tạo khác biệt hay không. Thứ hai, với cấu hình truy xuất tốt nhất đo được, chất lượng câu trả lời do mô hình ngôn ngữ sinh ra — đo bằng độ trung thực với ngữ cảnh, độ liên quan của câu trả lời và độ chính xác/độ phủ của trích dẫn — ở mức nào. Ba cấu hình truy xuất được chạy trên cùng một tập truy vấn, mỗi câu lưu tối đa Top-10 kết quả, và chỉ sau khi truy xuất kết thúc mới đối chiếu với nhãn chuẩn (\textbf{Mục \ref{sec:khung-danh-gia}}). Các chỉ số truy xuất là \textit{Precision@K}, \textit{Recall@K} và \textit{F1@K} tại $K\in\{1,5,10\}$; các chỉ số sinh câu trả lời được định nghĩa ở \textbf{Mục \ref{sec:ket-qua-sinh}}. Khả năng từ chối câu hỏi ngoài phạm vi chưa được đánh giá vì bộ câu hỏi hiện tại không có câu không thể trả lời.

\section{Môi trường thực nghiệm}\label{sec:moi-truong}

% [LOG EXPERIMENT_LOG.md mục 16.1]
\textbf{Thiết bị.} Toàn bộ thực nghiệm chạy trên một máy MacBook Pro, chip Apple M1 Pro (10 lõi: 8 lõi hiệu năng và 2 lõi tiết kiệm điện), 16GB bộ nhớ hợp nhất, hệ điều hành macOS 27.0 (bản dựng 26A428). Cả bốn lần chạy truy xuất đều dùng khung tăng tốc đồ họa của Apple (Metal Performance Shaders - MPS) để nạp mô hình: nhật ký hệ thống xác nhận trực tiếp hai lần chạy có bật bộ xếp hạng lại (nạp cả ba mô hình: biểu diễn văn bản, biểu diễn thị giác, xếp hạng lại); hai lần chạy còn lại (hệ thống cơ sở và cấu hình lai không xếp hạng lại) không có tệp nhật ký riêng, thiết bị MPS do tác giả xác nhận trực tiếp.

\bigbreak\noindent
\textbf{Phần mềm.} Môi trường Python 3.12.14 với PyTorch 2.14.0, Transformers 5.16.1, Sentence Transformers 6.0.1, ChromaDB 1.5.9, \texttt{rank-bm25} 0.2.2, \texttt{langchain-huggingface} 1.2.2, \texttt{faster-whisper} 1.2.1 và Docling 2.126.0. Các mô hình được nạp từ bộ nhớ đệm cục bộ ở chế độ ngoại tuyến (\texttt{HF\_HUB\_OFFLINE=1}, \texttt{TRANSFORMERS\_OFFLINE=1}). Mã băm phiên bản (revision) đọc trực tiếp từ thư mục bộ nhớ đệm cục bộ: Qwen3-Embedding-0.6B — \texttt{97b0c61}; Qwen3-VL-Embedding-2B — \texttt{9f2f7e7}; Qwen3-Reranker-0.6B — \texttt{e61197e}; Qwen3-1.7B (dùng cho cả sinh câu trả lời và chấm điểm giám khảo) — \texttt{70d244c}. Hai mô hình biểu diễn được nạp qua \texttt{SentenceTransformer} không chỉ định kiểu số nên dùng mặc định \texttt{float32}; bộ xếp hạng lại và mô hình sinh câu trả lời dùng \texttt{float16} trên MPS (\texttt{float32} trên CPU), theo mã nguồn (\texttt{app/retrieval/reranker.py}, \texttt{app/generation/service.py}).

\subsection{Kiểm định thống kê}\label{sec:kiem-dinh-v7}

Bộ câu hỏi v7 không có bản ghi trùng nội dung truy vấn (400 bản ghi, 400 cặp môn học/câu hỏi khác nhau), nên 400 bản ghi được coi là 400 đơn vị lấy mẫu độc lập, không cần tách lấy mẫu lại theo truy vấn và theo bản ghi. Chúng tôi tính khoảng tin cậy cho hiệu số giữa từng cặp hệ thống bằng phương pháp lấy mẫu lại ghép cặp \cite{efron1993bootstrap}, một hướng đã được so sánh với các kiểm định khác cho đánh giá truy xuất thông tin \cite{smucker2007comparison}: lấy mẫu có hoàn lại 400 bản ghi 10\,000 lần, giữ nguyên cặp kết quả của hai hệ thống trên cùng một bản ghi, tính hiệu số trung bình của từng chỉ số, và lấy khoảng giữa phân vị 2,5\% và 97,5\% làm khoảng tin cậy 95\%, với giá trị khởi tạo ngẫu nhiên (random seed) 20260930. \subsection{Kết quả sinh câu trả lời (generation-level)}\label{sec:ket-qua-sinh}

Để có kết quả đánh giá đầu-cuối (end-to-end evaluation) đầu tiên của khóa luận, với cấu hình truy xuất tốt nhất đo được (Lai + XHL), chúng tôi chạy toàn bộ luồng sinh câu trả lời (tăng cường câu hỏi, HyDE, truy xuất, xếp hạng lại, sinh câu trả lời bằng Qwen3-1.7B) trên 400 câu, chỉ dùng ngữ cảnh văn bản (không bật nhánh trả lời bằng ảnh), lấy tối đa 8 kết quả truy xuất làm ngữ cảnh cho mỗi câu (tham số \texttt{generation\_k=8}, khác với $K=10$ dùng ở phần truy xuất thuần). \section{Phân tích và thảo luận}\label{sec:thao-luan}

\textbf{Về giả thuyết đa phương thức.} Kết quả tách bạch được hai hiệu ứng riêng (\textbf{Mục \ref{sec:kiem-dinh-v7}}). Riêng việc thêm BM25 và truy xuất trên không gian thị giác, hợp nhất bằng RRF không trọng số nhưng chưa xếp hạng lại, không cải thiện so với truy xuất dày có lọc theo môn học — khoảng tin cậy chứa 0 ở $K=5,10$ và còn âm có ý nghĩa ở $K=1$. Ngược lại, thêm bước xếp hạng lại với đầu vào định dạng đúng theo mẫu hội thoại của mô hình mang lại cải thiện có ý nghĩa thống kê, nhất quán ở \textit{Precision}, \textit{Recall} và \textit{F1} tại $K=5$ và $K=10$. Nói cách khác, phần lớn lợi ích đo được trong thực nghiệm này đến từ bước xếp hạng lại được định dạng đúng, không phải từ việc hợp nhất nhiều danh sách truy xuất.

\bigbreak\noindent
\textbf{Về giới hạn cấu trúc của \textit{Precision@K}.} Do mỗi tệp chỉ được giữ tối đa 2 kết quả trong Top-K (lọc đa dạng theo tệp) và phần lớn câu hỏi chỉ có một hoặc hai nguồn bằng chứng, \textit{Precision@K} ở $K=5,10$ bị giới hạn cấu trúc: với một bản ghi chỉ có một bằng chứng, số kết quả liên quan tối đa là 2, nên \textit{Precision@10} khó vượt $0{,}2$. Vì vậy các giá trị \textit{Precision@5}, \textit{Precision@10} tuyệt đối thấp một phần là hệ quả của thiết lập truy xuất, không riêng của thuật toán, và nên đọc cùng \textit{Recall@K} và tỷ lệ trúng.

\bigbreak\noindent
\textbf{Về nhóm \textit{video\_grounded}.} Đây là nhóm yếu nhất ở cả ba hệ thống truy xuất và ở cả phần sinh câu trả lời (\textit{F1@10} cao nhất của nhóm này chỉ 0,0461, so với 0,0797–0,1839 ở các nhóm còn lại). Chúng tôi chưa có bằng chứng thực nghiệm về nguyên nhân; các giả thuyết có thể kiểm chứng gồm: cửa sổ phiên âm video (30 giây) quá thô so với yêu cầu khớp đúng khoảng thời gian của tiêu chí chấm điểm, hoặc câu hỏi nhóm này đòi hỏi mốc thời gian chính xác hơn mức mà truy xuất ở cấp đoạn phiên âm có thể cung cấp. Cần thí nghiệm riêng trước khi kết luận.

\bigbreak\noindent
\textbf{Về chất lượng sinh câu trả lời.} Cách đánh giá không dựa trên tài liệu tham chiếu ở đây dùng chính Qwen3-1.7B làm giám khảo — mô hình nhỏ, cùng loại với mô hình sinh câu trả lời — nên kết quả độ trung thực với ngữ cảnh/độ liên quan của câu trả lời mang hai rủi ro cần lưu ý khi diễn giải: (1) xu hướng tự thiên vị khi một mô hình chấm điểm sản phẩm của chính nó, chưa được kiểm chứng bằng giám khảo độc lập hay con người; (2) phân bố điểm gần như nhị phân (\textbf{Mục \ref{sec:ket-qua-sinh}}) cho thấy mô hình có thể không tuân theo đúng thang điểm liên tục $[0,1]$ như yêu cầu trong câu lệnh dẫn dắt. \textit{Citation precision} ở mức thấp (0,0847) phần lớn là hệ quả của mẫu số: mỗi câu đưa tối đa 8 kết quả vào ngữ cảnh trong khi số bằng chứng chuẩn trung bình mỗi câu nhỏ hơn nhiều, nên không nên so trực tiếp con số này với \textit{Precision@K} của phần truy xuất thuần (dùng $K=10$ và tiêu chí khớp khác).

\bigbreak\noindent
\textbf{Các mối đe dọa đến tính giá trị.}
\begin{itemize}
    \item Mỗi hệ thống chỉ được chạy một lần, trên một bộ câu hỏi từ một nguồn duy nhất; kiểm định thống kê so sánh 9 chỉ số và nhiều cặp hệ thống cùng lúc mà không hiệu chỉnh cho việc so sánh nhiều lần.
    \item Tiêu chí liên quan yêu cầu khớp chính xác số trang/số slide (và giao với khoảng thời gian đối với video), nên một kết quả ở vị trí liền kề không được tính là đúng.
    \item Mã môn học được truyền cho bộ truy xuất làm ngữ cảnh lọc; thực nghiệm chưa xét trường hợp không biết môn học.
    \item Phần sinh câu trả lời chỉ chạy với ngữ cảnh văn bản (chưa bật nhánh trả lời bằng ảnh), chỉ đo với một cấu hình truy xuất duy nhất, và giám khảo là mô hình nhỏ tự chấm sản phẩm của chính nó (nêu trên).
    \item Biến thể bật đồng thời HyDE và vòng lặp truy xuất lặp lại (\texttt{retrieve\_for\_eval}, đã có mã nguồn) chưa được đo cho cả phần truy xuất lẫn phần sinh câu trả lời.
\end{itemize}

\section{Ưu điểm và hạn chế}

\textbf{Ưu điểm}
\begin{itemize}
    \item \textit{Quy trình đánh giá không rò rỉ và có thể tái lập:} tệp truy vấn chỉ có ba trường, nhãn chuẩn chỉ được đọc sau khi truy xuất, tập truy vấn có mã băm và quá trình chạy có thể tiếp tục sau khi bị gián đoạn (\textbf{Mục \ref{sec:khung-danh-gia}}).
    \item \textit{Kho học liệu đầy đủ:} độ phủ 100\% đối với bộ câu hỏi đánh giá, không có câu nào bị loại vì thiếu nguồn.
    \item \textit{Tách bạch được hiệu ứng của từng thành phần:} ba cấu hình cho phép phân biệt hiệu ứng của việc hợp nhất đa phương thức (không rõ rệt) với hiệu ứng của xếp hạng lại (rõ rệt, có ý nghĩa thống kê).
    \item \textit{Đánh giá cả truy xuất lẫn sinh câu trả lời:} không chỉ đánh giá khả năng tìm bằng chứng, mà còn đo độ trung thực, độ liên quan và chất lượng trích dẫn của câu trả lời cuối cùng.
    \item \textit{Trích dẫn có vị trí:} câu trả lời và kết quả tìm kiếm gắn với tệp, số trang hoặc mốc thời gian, giúp người học kiểm chứng (\textbf{Mục \ref{sec:sinh}}).
    \item \textit{Báo cáo trung thực:} kết quả được báo cùng khoảng tin cậy, kể cả các so sánh cho kết quả âm hoặc không có ý nghĩa thống kê (hợp nhất đa phương thức khi chưa xếp hạng lại), không chỉ nêu kết quả tích cực.
\end{itemize}

\bigbreak\noindent
\textbf{Hạn chế}
\begin{itemize}
    \item Mỗi hệ thống chỉ chạy một lần; chưa hiệu chỉnh kiểm định cho việc so sánh nhiều chỉ số và nhiều cặp hệ thống.
    \item Giám khảo đánh giá sinh câu trả lời là mô hình nhỏ (Qwen3-1.7B) tự chấm sản phẩm của chính nó, phân bố điểm gần như nhị phân thay vì liên tục; chưa có đối chiếu với con người hoặc mô hình giám khảo độc lập.
    \item Nhóm câu hỏi \textit{video\_grounded} yếu ở mọi cấu hình đã đo, nguyên nhân chưa được kiểm chứng.
    \item Chưa đo biến thể HyDE + vòng lặp truy xuất lặp lại và biến thể trả lời bằng ảnh.
    \item Các hạn chế về thiết kế được nêu ở \textbf{Mục \ref{sec:han-che}}.
\end{itemize}
